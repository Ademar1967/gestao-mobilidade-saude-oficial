<script>
  var zoomAtual = 1;
  function aplicarZoom() {
    var wrapper = document.getElementById('folhasWrapper');
    var indicador = document.getElementById('zoomPercent');
    if (!wrapper) return;
    var zoomLimitado = Math.max(0.6, Math.min(1.6, zoomAtual));
    zoomAtual = zoomLimitado;
    wrapper.style.transform = 'scale(' + zoomAtual + ')';
    if (indicador) {
      indicador.textContent = Math.round(zoomAtual * 100) + '%';
    }
  }

  function ajustarZoom(delta) {
    zoomAtual += delta;
    aplicarZoom();
  }

  function restaurarZoom() {
    zoomAtual = 1;
    aplicarZoom();
  }

  function sairDaImpressao() {
    if (window.opener && !window.opener.closed) {
      window.close();
      return;
    }

    if (window.history.length > 1) {
      window.history.back();
      return;
    }

    window.location.href = '{% url 'transporte_pacientes:mapa_operacional' %}';
  }

  document.addEventListener('DOMContentLoaded', aplicarZoom);
</script>
